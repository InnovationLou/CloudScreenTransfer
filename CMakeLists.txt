cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(CloudScreenTransfer VERSION 0.2.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This release requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 NTDDI_VERSION=0x06010000)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/W4>)
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc)
add_library(cst_correct STATIC
    third_party/libcorrect/src/reed-solomon/polynomial.c
    third_party/libcorrect/src/reed-solomon/reed-solomon.c
    third_party/libcorrect/src/reed-solomon/encode.c
    third_party/libcorrect/src/reed-solomon/decode.c)
target_include_directories(cst_correct PUBLIC third_party/libcorrect/include)
add_library(cst_core STATIC src/protocol.cpp src/frame_codec.cpp src/file_transfer.cpp src/image_runtime.cpp src/encoding.cpp src/settings.cpp)
target_include_directories(cst_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(cst_core PUBLIC cst_correct ${OpenCV_LIBS} advapi32 shell32)
add_library(cst_win7 OBJECT src/win7_compat.cpp)
add_executable(CloudScreenTransfer WIN32 src/gui.cpp src/windows_channel.cpp src/diagnostics.cpp src/advanced_settings.cpp $<TARGET_OBJECTS:cst_win7>)
target_link_libraries(CloudScreenTransfer PRIVATE cst_core user32 gdi32 shell32 comdlg32 comctl32 ole32 imm32)
enable_testing()
add_executable(cst_tests tests/test_main.cpp tests/image_runtime_tests.cpp tests/protocol_tests.cpp tests/frame_codec_tests.cpp tests/file_transfer_tests.cpp tests/end_to_end_tests.cpp $<TARGET_OBJECTS:cst_win7>)
target_link_libraries(cst_tests PRIVATE cst_core)
add_test(NAME transfer_correctness COMMAND cst_tests)
add_test(NAME gui_smoke COMMAND CloudScreenTransfer --smoke-test)
add_test(NAME sender_restart_smoke COMMAND CloudScreenTransfer --sender-smoke-test)
add_test(NAME gui_settings_restore COMMAND CloudScreenTransfer --settings-smoke-test)
set_tests_properties(gui_smoke sender_restart_smoke gui_settings_restore PROPERTIES TIMEOUT 60)
install(TARGETS CloudScreenTransfer RUNTIME DESTINATION .)
